  <script>
    (function() {
      const copyButton = document.getElementById('copyButton');
      const linkToCopy = 'https://raw.githubusercontent.com/FR1nter/FRinter/refs/heads/main/FRinter.txt';

      const iconCopy = `<svg class="icon" viewBox="0 0 24 24" aria-hidden="true"><rect x="9" y="9" width="11" height="11" rx="2"/><path d="M5 15 H4 a2 2 0 0 1 -2 -2 V4 a2 2 0 0 1 2 -2 h9 a2 2 0 0 1 2 2 v1"/></svg>`;
      const iconCheck = `<svg class="icon" viewBox="0 0 24 24" aria-hidden="true"><path d="M20 6 L9 17 L4 12"/></svg>`;

      async function copyToClipboard() {
        try {
          await navigator.clipboard.writeText(linkToCopy);
          showCopiedState();
        } catch (err) {
          fallbackCopy(linkToCopy);
        }
      }

      function showCopiedState() {
        copyButton.innerHTML = `${iconCheck} Скопировано`;
        copyButton.style.background = '#2e5e2e';
        copyButton.style.borderColor = '#4a8a4a';

        setTimeout(() => {
          copyButton.innerHTML = `${iconCopy} Скопировать`;
          copyButton.style.background = '';
          copyButton.style.borderColor = '';
        }, 1500);
      }

      function fallbackCopy(text) {
        const textArea = document.createElement('textarea');
        textArea.value = text;
        textArea.style.position = 'fixed';
        textArea.style.opacity = '0';
        textArea.style.left = '-9999px';
        document.body.appendChild(textArea);
        textArea.select();
        try {
          document.execCommand('copy');
          showCopiedState();
        } catch (e) {
          alert('Не удалось скопировать ссылку. Пожалуйста, скопируйте вручную.');
        } finally {
          document.body.removeChild(textArea);
        }
      }

      copyButton.addEventListener('click', copyToClipboard);
    })();
  </script>
